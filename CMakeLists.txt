cmake_minimum_required(VERSION 3.16)
project(edgeguard_adaptive LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(adaptive_router cpp/src/router.cpp)
target_include_directories(adaptive_router PUBLIC cpp/include)
target_compile_options(adaptive_router PRIVATE -Wall -Wextra -Wpedantic)
add_executable(adaptive_demo cpp/src/main.cpp)
target_link_libraries(adaptive_demo PRIVATE adaptive_router)
enable_testing()
add_executable(adaptive_router_test cpp/tests/router_test.cpp)
target_link_libraries(adaptive_router_test PRIVATE adaptive_router)
add_test(NAME adaptive_router_test COMMAND adaptive_router_test)
